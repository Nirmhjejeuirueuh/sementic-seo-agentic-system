# --- Neo4j -----------------------------------------------------------
neo4j>=5.26,<6

# --- LangGraph -------------------------------------------------------
# We pin ONLY langgraph and let pip resolve langgraph-checkpoint and
# langgraph-prebuilt from langgraph's own dependency metadata.
#
# Why: the previous version of this file pinned all three to 0.2.60.
# langgraph-prebuilt never had a 0.2.x release, so the set was
# unresolvable and `pip install -r requirements.txt` failed outright.
# Guessing matching version numbers by hand is how that happens.
# langgraph already declares which versions of its siblings it works
# with -- trust that instead of guessing.
langgraph>=1.2,<1.3
